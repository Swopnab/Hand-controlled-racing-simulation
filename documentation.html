1. Executive Summary

CyberSteer is a browser-based, AI-driven highway racing and obstacle dodging game. It combines real-time client-side computer vision with a high-performance 60 FPS HTML5 2D Canvas game loop.

Players can steer their cybernetic racing car either using their physical hands tracked live via webcam (using Google MediaPipe Hands ML models) or standard keyboard hotkeys (A/D or Arrow keys).

3. Key Modules & Technical Implementation

3.1. Computer Vision & Gesture Steering (handControl.js)

  • MediaPipe Hands Integration: Initializes an in-browser ML model targeting 21 hand landmarks running asynchronously inside requestAnimationFrame.
  • Palm Center Calculation: Extracts Landmark 9 (Middle Finger MCP), normalized to [0, 1] horizontal coordinates.
  • Mirror Space Transformation: Converts camera coordinates rawX = 1 - palmCenter.x so moving hand right moves the car right from the user's natural perspective.
  • Hysteresis Zone Thresholding: Divides screen into LEFT (< 0.38), CENTER (0.38 - 0.62), and RIGHT (> 0.62). Implements a ±0.03 hysteresis buffer to eliminate jitter near boundaries.
  • Dynamic Neon Overlay: Visualizes the hand skeleton with glowing cyan lines, joint nodes, and palm reticle directly over the live camera feed.

3.2. Game Engine & Speed Progression (game.js)

  • State Machine: Controls MENU, PLAYING, PAUSED, and GAMEOVER states with smooth modal transitions.
  • Steering Interpolation (Lerp): Smooths lane changes using linear interpolation player.x += (targetX - player.x) * 0.18.
  • Speed Progression: Starts at 50 mph and progresses through 75 mph, 90 mph, capping strictly at 100 mph. Continuous lerp makes speed transitions feel realistic.
  • Four-Wheel Obstacle Variety: Procedurally spawns 4 distinct vehicle classes (Cars, SUVs, Pickups, and Semi-Trucks) with custom pixel silhouettes, wheels, and taillights.
  • Particle Engine: Spawns 30 velocity-driven expanding debris particles with alpha decay on collisions.
  • AABB Collision Detection: Bounding box collision with customizable padding for forgiving yet precise collision boundaries.

3.3. Responsive Arcade UI & Layout (index.html & style.css)

  • Dual-Panel Split Screen: Aligns the left game window and right live camera panel at 470px × 630px side-by-side.
  • Context-Aware Right Panel: Displays game instructions & controls during Menu mode, and dynamically expands to a full-screen live webcam feed during gameplay.
  • Pixel Font & Hierarchy: Imports Google Font Press Start 2P with gold headings (#ffd700), white body text (#ffffff), and cyan neon button accents (#00f2fe).

// Game Engine Logic for Hand-Controlled Car Dodger

const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');

// Game State Constants
const STATES = {
    MENU: 'MENU',
    PLAYING: 'PLAYING',
    PAUSED: 'PAUSED',
    GAMEOVER: 'GAMEOVER'
};

let gameState = STATES.MENU;

// Speed Progression (50 -> 75 -> 90 -> 100 mph max)
const SPEED_STAGES = [50, 75, 90, 100];
let currentSpeedMph = 50;
let targetSpeedMph = 50;
let speedStage = 0;
let stageTimer = 0;

// High Score management
let highScore = parseInt(localStorage.getItem('carDodgerHighScore') || '0');

// Input Settings
let controlMode = 'KEYBOARD'; // 'KEYBOARD' or 'HAND'

// Road Design (Scaled for 470x630 canvas)
const ROAD = {
    x: 75,
    width: 320,
    lanes: [128, 235, 342], // Center X of each lane
    laneWidth: 107,
    speed: 5,
    maxSpeed: 10,
    offset: 0
};

// Player Object (Slightly smaller, centered)
const player = {
    lane: 1, // Start in Center lane
    x: ROAD.lanes[1] - 16,
    targetX: ROAD.lanes[1] - 16,
    y: 510,
    width: 32,
    height: 62,
    color: '#00f2fe',
    lerpSpeed: 0.18, // Smoothness of steering
    score: 0,
    reset() {
        this.lane = 1;
        this.targetX = ROAD.lanes[1] - this.width / 2;
        this.x = this.targetX;
        this.y = 510;
        this.score = 0;
    }
};

// Enemies List
let enemies = [];
let enemySpawnTimer = 0;
let enemySpawnInterval = 120;
const minSpawnInterval = 45;

// Particle Explosions
let particles = [];

// Keyboard listener setup
const keys = {};
window.addEventListener('keydown', (e) => {
    keys[e.code] = true;
    
    // State-independent hotkeys
    if (e.code === 'KeyP' || e.code === 'Escape') {
        if (gameState === STATES.PLAYING) {
            gameState = STATES.PAUSED;
            showUIOverlay('paused-screen');
        } else if (gameState === STATES.PAUSED) {
            gameState = STATES.PLAYING;
            hideUIOverlays();
        }
    }
    
    // Quick steering tests (Keyboard fallback)
    if (gameState === STATES.PLAYING && controlMode === 'KEYBOARD') {
        if (e.code === 'ArrowLeft' || e.code === 'KeyA') {
            steerLeft();
        }
        if (e.code === 'ArrowRight' || e.code === 'KeyD') {
            steerRight();
        }
    }
});
window.addEventListener('keyup', (e) => {
    keys[e.code] = false;
});

// Steering logic
function steerLeft() {
    if (player.lane > 0) {
        player.lane--;
        player.targetX = ROAD.lanes[player.lane] - player.width / 2;
    }
}

function steerRight() {
    if (player.lane < 2) {
        player.lane++;
        player.targetX = ROAD.lanes[player.lane] - player.width / 2;
    }
}

// Map detected hand zone to lane selection
function handleHandSteering() {
    if (window.handControl && window.handControl.isActive && window.handControl.handDetected) {
        const zone = window.handControl.currentZone;
        let targetLane = player.lane;
        
        if (zone === 'LEFT') {
            targetLane = 0;
        } else if (zone === 'CENTER') {
            targetLane = 1;
        } else if (zone === 'RIGHT') {
            targetLane = 2;
        }

        if (targetLane !== player.lane) {
            player.lane = targetLane;
            player.targetX = ROAD.lanes[player.lane] - player.width / 2;
        }
    }
}

// Enemy Spawning and management (Cars, SUVs, Pickups, Trucks - No motorcycles)
function spawnEnemy() {
    const lane = Math.floor(Math.random() * 3);
    
    const types = ['CAR', 'SUV', 'PICKUP', 'TRUCK'];
    const type = types[Math.floor(Math.random() * types.length)];
    
    let width, height;
    if (type === 'CAR') {
        width = 32;
        height = 62;
    } else if (type === 'SUV') {
        width = 36;
        height = 68;
    } else if (type === 'PICKUP') {
        width = 36;
        height = 78;
    } else { // TRUCK
        width = 42;
        height = 105;
    }
    const x = ROAD.lanes[lane] - width / 2;
    
    // Randomize colors
    const colors = ['#ef4444', '#a855f7', '#f59e0b', '#3b82f6', '#10b981', '#ec4899'];
    const color = colors[Math.floor(Math.random() * colors.length)];
    
    const speedOffset = (type === 'TRUCK') ? -1 : ((type === 'CAR') ? 0.5 : 0);

    enemies.push({
        x: x,
        y: -height - 20,
        width: width,
        height: height,
        color: color,
        speed: ROAD.speed + speedOffset + (Math.random() * 1.5 - 0.5),
        lane: lane,
        type: type
    });
}

// Particle System for collision explosions
function createExplosion(x, y, color) {
    const count = 30;
    for (let i = 0; i < count; i++) {
        particles.push({
            x: x,
            y: y,
            vx: (Math.random() - 0.5) * 8,
            vy: (Math.random() - 0.5) * 8,
            size: Math.random() * 4 + 2,
            alpha: 1,
            decay: Math.random() * 0.03 + 0.01,
            color: color
        });
    }
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.alpha -= p.decay;
        if (p.alpha <= 0) {
            particles.splice(i, 1);
        }
    }
}

function drawParticles() {
    particles.forEach(p => {
        ctx.save();
        ctx.globalAlpha = p.alpha;
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
    });
}

// Collision helper using Axis-Aligned Bounding Box (AABB)
function checkCollision(r1, r2) {
    const padding = 3;
    return r1.x + padding < r2.x + r2.width - padding &&
           r1.x + r1.width - padding > r2.x + padding &&
           r1.y + padding < r2.y + r2.height - padding &&
           r1.y + r1.height - padding > r2.y + padding;
}

// Update loop logic
function updateGame() {
    if (gameState !== STATES.PLAYING) return;

    // 1. Move road to simulate speed
    ROAD.offset = (ROAD.offset + ROAD.speed) % 70;

    // 2. Gradual Speed Progression (50 -> 75 -> 90 -> 100 mph max)
    if (speedStage < 3) {
        stageTimer += 1 / 60;
        if (stageTimer >= 15) {
            speedStage++;
            stageTimer = 0;
            targetSpeedMph = SPEED_STAGES[speedStage];
        }
    }

    // Smooth transition between speed levels
    currentSpeedMph += (targetSpeedMph - currentSpeedMph) * 0.015;
    if (currentSpeedMph > 100) currentSpeedMph = 100;

    // Map current mph to road speed (50 mph = 5, 100 mph = 10)
    ROAD.speed = (currentSpeedMph / 50) * 5;

    // Dynamic spawn interval based on score
    enemySpawnInterval = Math.max(minSpawnInterval, 120 - Math.floor(player.score / 5));

    // 3. Interpolate player steering
    player.x += (player.targetX - player.x) * player.lerpSpeed;

    // 4. Hand gesture steering
    if (controlMode === 'HAND') {
        handleHandSteering();
    }

    // 5. Spawn enemies
    enemySpawnTimer++;
    if (enemySpawnTimer >= enemySpawnInterval) {
        spawnEnemy();
        enemySpawnTimer = 0;
    }

    // 6. Update enemies
    for (let i = enemies.length - 1; i >= 0; i--) {
        const enemy = enemies[i];
        enemy.y += enemy.speed;

        // Collision Check
        if (checkCollision(player, enemy)) {
            triggerGameOver(enemy);
            return;
        }

        // Delete passed enemies, increment score
        if (enemy.y > canvas.height) {
            enemies.splice(i, 1);
            player.score++;
            updateScoreHUD();
        }
    }

    // Update HUD speed continuously for smooth numbers
    updateScoreHUD();

    // Update particles
    updateParticles();
}

// Handle Game HUD rendering
function updateScoreHUD() {
    document.getElementById('hud-score').textContent = player.score;
    document.getElementById('hud-speed').textContent = Math.round(currentSpeedMph) + ' mph';
}

// End game logic
function triggerGameOver(collidedEnemy) {
    gameState = STATES.GAMEOVER;
    createExplosion(player.x + player.width / 2, player.y + player.height / 2, player.color);
    createExplosion(collidedEnemy.x + collidedEnemy.width / 2, collidedEnemy.y + collidedEnemy.height / 2, collidedEnemy.color);

    // Save High Score
    if (player.score > highScore) {
        highScore = player.score;
        localStorage.setItem('carDodgerHighScore', highScore.toString());
    }

    setTimeout(() => {
        document.getElementById('over-score').textContent = player.score;
        document.getElementById('over-highscore').textContent = highScore;
        showUIOverlay('gameover-screen');
    }, 1000);
}

// Master Render loop
function draw() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    if (gameState === STATES.PLAYING || gameState === STATES.PAUSED || gameState === STATES.GAMEOVER) {
        drawRoad();
        
        // Draw enemies
        enemies.forEach(drawEnemyCar);

        // Draw player (skip on gameover if exploded)
        if (gameState !== STATES.GAMEOVER || particles.length > 0) {
            drawPlayerCar(player.x, player.y, player.width, player.height);
        }

        // Draw particle explosion
        drawParticles();
    } else {
        // Draw clean road under menu for visuals
        drawRoad();
        ctx.fillStyle = 'rgba(10, 15, 26, 0.6)';
        ctx.fillRect(0, 0, canvas.width, canvas.height);
    }
}

}

// Hand Tracking & Gesture Steering Module using MediaPipe Hands

const handControl = {
    isActive: false,
    handDetected: false,
    currentZone: 'CENTER', // 'LEFT', 'CENTER', 'RIGHT'
    rawX: 0.5, // Normalized X position (0 to 1, mirrored for player perspective)
    rawY: 0.5,
    
    // Elements
    videoElement: null,
    canvasElement: null,
    canvasCtx: null,
    hands: null,
    camera: null,
    onFrame: null,

    // Thresholds with hysteresis
    thresholds: {
        leftBound: 0.38,
        rightBound: 0.62,
        hysteresis: 0.03
    },

    // Initialize webcam and MediaPipe Hands
    async init(videoEl, overlayCanvasEl) {
        this.videoElement = videoEl;
        this.canvasElement = overlayCanvasEl;
        this.canvasCtx = overlayCanvasEl.getContext('2d');

        if (typeof Hands === 'undefined') {
            console.error('MediaPipe Hands library is not loaded. Ensure CDN scripts are loaded in index.html.');
            return false;
        }

        try {
            this.hands = new Hands({
                locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/hands/${file}`
            });

            this.hands.setOptions({
                maxNumHands: 1,
                modelComplexity: 1,
                minDetectionConfidence: 0.6,
                minTrackingConfidence: 0.6
            });

            this.hands.onResults((results) => this.onResults(results));

            this.camera = new Camera(this.videoElement, {
                onFrame: async () => {
                    if (this.isActive) {
                        await this.hands.send({ image: this.videoElement });
                    }
                },
                width: 320,
                height: 240
            });

            return true;
        } catch (err) {
            console.error('Error starting MediaPipe Hands:', err);
            return false;
        }
    },

    start() {
        if (this.camera) {
            this.isActive = true;
            this.camera.start();
        }
    },

    stop() {
        this.isActive = false;
        if (this.camera) {
            this.camera.stop();
        }
    },

    // Process Hand Landmarks
    onResults(results) {
        const ctx = this.canvasCtx;
        const w = this.canvasElement.width;
        const h = this.canvasElement.height;

        ctx.clearRect(0, 0, w, h);

        // Draw camera image mirrored
        ctx.save();
        ctx.translate(w, 0);
        ctx.scale(-1, 1);
        ctx.drawImage(results.image, 0, 0, w, h);
        ctx.restore();

        // Draw visual zone guides (Left, Center, Right)
        this.drawZoneGuides(ctx, w, h);

        if (results.multiHandLandmarks && results.multiHandLandmarks.length > 0) {
            this.handDetected = true;
            const landmarks = results.multiHandLandmarks[0];

            // In mirrored space: x_mirrored = 1 - x_raw
            // Landmark 9 is the Middle MCP (palm center), landmark 0 is wrist
            const palmCenter = landmarks[9];
            this.rawX = 1 - palmCenter.x;
            this.rawY = palmCenter.y;

            // Determine zone
            this.currentZone = this.calculateZone(this.rawX, this.currentZone);

            // Draw glowing hand skeleton in mirrored orientation
            this.drawHandSkeleton(ctx, landmarks, w, h);
        } else {
            this.handDetected = false;
            // Retain last zone or default to CENTER when hand removed
            this.currentZone = 'CENTER';
        }

        if (this.onFrame) {
            this.onFrame();
        }
    },

    // Calculate zone with hysteresis to prevent twitching near borders
    calculateZone(x, prevZone) {
        const { leftBound, rightBound, hysteresis } = this.thresholds;

        if (prevZone === 'LEFT') {
            if (x > leftBound + hysteresis) {
                return x > rightBound ? 'RIGHT' : 'CENTER';
            }
            return 'LEFT';
        } else if (prevZone === 'RIGHT') {
            if (x < rightBound - hysteresis) {
                return x < leftBound ? 'LEFT' : 'CENTER';
            }
            return 'RIGHT';
        } else { // prevZone === 'CENTER'
            if (x < leftBound - hysteresis) return 'LEFT';
            if (x > rightBound + hysteresis) return 'RIGHT';
            return 'CENTER';
        }
    },

    // Draw Left/Center/Right zone overlays on the camera preview
    drawZoneGuides(ctx, w, h) {
        const leftX = w * this.thresholds.leftBound;
        const rightX = w * this.thresholds.rightBound;

        // Highlight active zone
        ctx.fillStyle = 'rgba(0, 242, 254, 0.15)';
        if (this.currentZone === 'LEFT' && this.handDetected) {
            ctx.fillRect(0, 0, leftX, h);
        } else if (this.currentZone === 'CENTER' && this.handDetected) {
            ctx.fillRect(leftX, 0, rightX - leftX, h);
        } else if (this.currentZone === 'RIGHT' && this.handDetected) {
            ctx.fillRect(rightX, 0, w - rightX, h);
        }

        // Zone dividing lines
        ctx.strokeStyle = 'rgba(255, 255, 255, 0.25)';
        ctx.lineWidth = 1.5;
        ctx.setLineDash([4, 4]);

        ctx.beginPath();
        ctx.moveTo(leftX, 0);
        ctx.lineTo(leftX, h);
        ctx.moveTo(rightX, 0);
        ctx.lineTo(rightX, h);
        ctx.stroke();
        ctx.setLineDash([]);

        // Zone Text Labels
        ctx.font = "bold 10px 'Segoe UI', sans-serif";
        ctx.textAlign = 'center';
        ctx.fillStyle = this.currentZone === 'LEFT' && this.handDetected ? '#00f2fe' : 'rgba(255, 255, 255, 0.5)';
        ctx.fillText('LEFT', leftX / 2, 20);

        ctx.fillStyle = this.currentZone === 'CENTER' && this.handDetected ? '#00f2fe' : 'rgba(255, 255, 255, 0.5)';
        ctx.fillText('CENTER', (leftX + rightX) / 2, 20);

        ctx.fillStyle = this.currentZone === 'RIGHT' && this.handDetected ? '#00f2fe' : 'rgba(255, 255, 255, 0.5)';
        ctx.fillText('RIGHT', (rightX + w) / 2, 20);
    },

    // Draw futuristic neon cyber hand skeleton
    drawHandSkeleton(ctx, landmarks, w, h) {
        // Mirrored coordinate helper
        const getPt = (idx) => ({
            x: (1 - landmarks[idx].x) * w,
            y: landmarks[idx].y * h
        });

        // Skeleton connections
        const connections = [
            // Palm
            [0, 1], [0, 5], [5, 9], [9, 13], [13, 17], [0, 17],
            // Thumb
            [1, 2], [2, 3], [3, 4],
            // Index
            [5, 6], [6, 7], [7, 8],
            // Middle
            [9, 10], [10, 11], [11, 12],
            // Ring
            [13, 14], [14, 15], [15, 16],
            // Pinky
            [17, 18], [18, 19], [19, 20]
        ];

        // Draw glowing lines
        ctx.save();
        ctx.shadowBlur = 8;
        ctx.shadowColor = '#00f2fe';
        ctx.strokeStyle = 'rgba(0, 242, 254, 0.8)';
        ctx.lineWidth = 2.5;

        connections.forEach(([i, j]) => {
            const p1 = getPt(i);
            const p2 = getPt(j);
            ctx.beginPath();
            ctx.moveTo(p1.x, p1.y);
            ctx.lineTo(p2.x, p2.y);
            ctx.stroke();
        });

        // Draw joint nodes
        ctx.fillStyle = '#ffffff';
        landmarks.forEach((_, idx) => {
            const p = getPt(idx);
            const isFingertip = [4, 8, 12, 16, 20].includes(idx);
            ctx.beginPath();
            ctx.arc(p.x, p.y, isFingertip ? 4.5 : 2.5, 0, Math.PI * 2);
            ctx.fillStyle = isFingertip ? '#00f2fe' : '#ffffff';
            ctx.fill();
        });

        // Palm centroid reticle
        const centerPt = getPt(9);
        ctx.strokeStyle = '#ef4444';
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.arc(centerPt.x, centerPt.y, 8, 0, Math.PI * 2);
        ctx.stroke();

        ctx.restore();
    }
};

window.handControl = handControl;
